=====================================================================
БИЛЕТ № 1. Отказоустойчивая пара DNS-серверов BIND9 на РедОС
(по ЛР ОРСОС №2 «Развёртывание отказоустойчивой пары DNS-серверов»)
=====================================================================

ЗАДАНИЕ ИЗ БИЛЕТА
- Настроить сервер 1 (Primary) и сервер 2 (Secondary): сеть, адреса,
  конфигурационные файлы, TSIG-ключ, /etc/named.conf.
- Проверить работу не менее чем ДВУМЯ способами.
- Остановить Primary и показать, что DNS продолжает работать.

СХЕМА (как в лабе)
  ns1.lab1.org  192.168.122.55  - Primary  (хранит ОРИГИНАЛ зоны)
  ns2.lab1.org  192.168.122.25  - Secondary (хранит КОПИЮ, берёт её с ns1)
  клиент        192.168.122.246
  шлюз          192.168.122.1
  зона          lab1.org + обратная 122.168.192.in-addr.arpa

ИДЕЯ В ДВУХ СЛОВАХ
Если зона лежит на одном сервере и он упал - клиенты перестают находить
имена. Поэтому ставим второй сервер: он копирует зону с первого
(зонный трансфер, AXFR/IXFR) и отвечает сам, когда первый недоступен.
TSIG-ключ - общий секрет двух серверов: без него никто чужой не скачает
зону и не выдаст себя за наш Primary.

Все команды - от root (sudo -i). В лабе было через sudo - это то же самое.
Имя подключения смотреть командой: nmcli con show


---------------------------------------------------------------------
ШАГ 1. Установка BIND                                [ГДЕ: ns1 И ns2]
---------------------------------------------------------------------
dnf install bind bind-utils
#  bind       - сам DNS-сервер (служба named)
#  bind-utils - dig, nslookup, host (для проверок)
#  После установки появляется пользователь named - от него работает служба.

systemctl enable named
#  ЗАЧЕМ: служба будет стартовать при загрузке системы.

ls -la /var/named/
#  ЗАЧЕМ: посмотреть ТОЧНЫЕ имена служебных файлов (named.ca,
#  named.localhost, named.loopback). Их потом пишем в named.conf.
#  Если указать несуществующий файл - named не запустится (file not found).


---------------------------------------------------------------------
ШАГ 2. Статический адрес и имя                       [ГДЕ: ns1 И ns2]
---------------------------------------------------------------------
ЗАЧЕМ: адреса DNS-серверов прописаны в зонах, в allow-transfer и у
клиентов. Если адрес сменится по DHCP - всё сломается.

# --- на ns1 ---
nmcli con mod "Проводное подключение 1" ipv4.addresses "192.168.122.55/24"
nmcli con mod "Проводное подключение 1" ipv4.gateway "192.168.122.1"
nmcli con mod "Проводное подключение 1" ipv4.dns "192.168.122.55"
nmcli con mod "Проводное подключение 1" ipv4.method manual
#  addresses - адрес/маска; gateway - шлюз; dns - DNS-сервер (сам ns1);
#  method manual - ВЫКЛЮЧАЕТ DHCP, иначе ручные настройки не применятся.
nmcli con down "Проводное подключение 1"; nmcli con up "Проводное подключение 1"
#  ЗАЧЕМ: переподключение, чтобы новые настройки вступили в силу.

hostnamectl set-hostname ns1.lab1.org
echo "192.168.122.55 ns1.lab1.org ns1" >> /etc/hosts
echo "192.168.122.25 ns2.lab1.org ns2" >> /etc/hosts
#  ЗАЧЕМ /etc/hosts: серверы находят друг друга по имени даже без DNS.
#  (Если работаешь не от root, а через sudo: sudo sh -c 'echo ... >> /etc/hosts',
#   потому что >> выполняет оболочка обычного пользователя, а не sudo.)

# --- на ns2 --- то же самое, НО адрес 192.168.122.25/24 и имя ns2.lab1.org
#  ipv4.dns на ns2 тоже ставим 192.168.122.55 (оригинал зоны на ns1).

# --- на обоих ---
firewall-cmd --permanent --add-service=dns
firewall-cmd --reload
#  ЗАЧЕМ: открыть порт 53 UDP и TCP. TCP обязателен для зонного трансфера!


---------------------------------------------------------------------
ШАГ 3. TSIG-ключ                                           [ГДЕ: ns1]
---------------------------------------------------------------------
rndc-confgen -a -c /etc/named.transfer.key -k transfer-key -A hmac-sha256
#  -a  записать только ключ (без полного конфига rndc)
#  -c  в какой файл
#  -k  имя ключа  (transfer-key - это имя потом пишем в named.conf)
#  -A  алгоритм hmac-sha256

chown root:named /etc/named.transfer.key
chmod 640 /etc/named.transfer.key
#  ЗАЧЕМ: named (группа) может ЧИТАТЬ ключ, остальные - нет. Ключ секретный.

cat /etc/named.transfer.key
#  Внутри: key "transfer-key" { algorithm hmac-sha256; secret "...base64..."; };


---------------------------------------------------------------------
ШАГ 4. /etc/named.conf на Primary                          [ГДЕ: ns1]
---------------------------------------------------------------------
vim /etc/named.conf     (стираем/правим содержимое до такого вида)

options {
    listen-on port 53 { any; };          # слушать IPv4 на всех интерфейсах
    listen-on-v6 port 53 { none; };      # IPv6 не нужен
    directory "/var/named";              # рабочий каталог - отсюда берутся file "..."
    dump-file "/var/named/data/cache_dump.db";
    statistics-file "/var/named/data/named_stats.txt";
    memstatistics-file "/var/named/data/named_mem_stats.txt";
    allow-query { any; };                # отвечать всем
    allow-recursion { 192.168.122.0/24; };   # рекурсия только своей сети
    allow-transfer { key "transfer-key"; 192.168.122.25; };  # зону отдаём только ns2 и только с ключом
    notify yes;                          # сообщать Secondary об изменении зоны
    also-notify { 192.168.122.25; };     # кому именно сообщать
    dnssec-validation no;                # учебная сеть, DNSSEC не проверяем
    version "DNS Server";                # прячем настоящую версию BIND
};

include "/etc/named.transfer.key";       # подключаем ключ из отдельного файла

zone "." IN { type hint; file "named.ca"; };      # корневые серверы
zone "localhost" IN { type master; file "named.localhost"; allow-update { none; }; };
zone "127.in-addr.arpa" IN { type master; file "named.loopback"; allow-update { none; }; };

zone "lab1.org" IN {                     # ПРЯМАЯ зона: имя -> IP
    type master;                         # master = оригинал хранится здесь
    file "lab1.org.zone";
    allow-transfer { key "transfer-key"; 192.168.122.25; };
    allow-update { none; };
};
zone "122.168.192.in-addr.arpa" IN {     # ОБРАТНАЯ зона: IP -> имя
    type master;                         # (октеты сети задом наперёд)
    file "192.168.122.zone";
    allow-transfer { key "transfer-key"; 192.168.122.25; };
    allow-update { none; };
};

logging { channel default_debug { file "data/named.run"; severity dynamic; }; };
#  журнал отладки - пригодится для проверки трансфера (named.run)


---------------------------------------------------------------------
ШАГ 5. Файлы зон                                           [ГДЕ: ns1]
---------------------------------------------------------------------
vim /var/named/lab1.org.zone

$TTL 86400
@ IN SOA ns1.lab1.org. admin.lab1.org. (
    2024012002 ; Serial  - ВЕРСИЯ зоны. Поменял зону -> увеличь Serial!
    3600       ; Refresh - как часто ns2 проверяет, не изменилась ли зона
    1800       ; Retry   - повтор, если ns1 не ответил
    604800     ; Expire  - через сколько ns2 перестанет отвечать без связи с ns1
    86400 )    ; Minimum TTL
    IN NS ns1.lab1.org.        ; ДВЕ записи NS - оба сервера авторитетны
    IN NS ns2.lab1.org.
@    IN A 192.168.122.55
ns1  IN A 192.168.122.55
ns2  IN A 192.168.122.25
www  IN A 192.168.122.50
ftp  IN A 192.168.122.51
mail IN A 192.168.122.52
test IN A 192.168.122.150
web   IN CNAME www
files IN CNAME ftp
@ IN MX 10 mail.lab1.org.
#  ТОЧКА в конце полных имён обязательна! Без неё BIND допишет зону:
#  получится ns1.lab1.org.lab1.org.
#  admin.lab1.org. = почта admin@lab1.org (вместо @ пишется точка).

vim /var/named/192.168.122.zone

$TTL 86400
@ IN SOA ns1.lab1.org. admin.lab1.org. (
    2024012001 3600 1800 604800 86400 )
    IN NS ns1.lab1.org.
    IN NS ns2.lab1.org.
55  IN PTR ns1.lab1.org.
25  IN PTR ns2.lab1.org.
50  IN PTR www.lab1.org.
51  IN PTR ftp.lab1.org.
52  IN PTR mail.lab1.org.
150 IN PTR test.lab1.org.
#  В обратной зоне пишем только последний октет адреса.

chown named:named /var/named/lab1.org.zone /var/named/192.168.122.zone
#  ЗАЧЕМ: файлы создавали от root, а читать их должна служба named.


---------------------------------------------------------------------
ШАГ 6. Проверка и запуск Primary                           [ГДЕ: ns1]
---------------------------------------------------------------------
named-checkconf -z /etc/named.conf
#  проверка синтаксиса named.conf; -z дополнительно загружает все master-зоны
named-checkzone lab1.org /var/named/lab1.org.zone
named-checkzone 122.168.192.in-addr.arpa /var/named/192.168.122.zone
#  должно быть "OK" и serial зоны

systemctl restart named
systemctl status named
#  ждём active (running), в журнале "all zones loaded", "sending notifies"
#  (sending notifies = ns1 сообщил ns2 о зонах)


---------------------------------------------------------------------
ШАГ 7. Перенос ключа на Secondary                  [ГДЕ: ns1 -> ns2]
---------------------------------------------------------------------
ЗАЧЕМ: ключ на ns2 должен совпадать байт в байт, иначе
"tsig verify failure". Поэтому НЕ перепечатываем, а копируем.

# на ns1:
scp /etc/named.transfer.key user@192.168.122.25:/tmp/
#  user - имя твоего обычного пользователя на ns2

# на ns2:
mv /tmp/named.transfer.key /etc/named.transfer.key
chown root:named /etc/named.transfer.key
chmod 640 /etc/named.transfer.key
restorecon -v /etc/named.transfer.key
#  ЗАЧЕМ restorecon: при mv файл сохранил метку SELinux от /tmp (user_tmp_t).
#  restorecon ставит правильную etc_t. Без этого SELinux не даст named
#  прочитать ключ, и трансфер не пойдёт.


---------------------------------------------------------------------
ШАГ 8. /etc/named.conf на Secondary                        [ГДЕ: ns2]
---------------------------------------------------------------------
vim /etc/named.conf

options - КАК НА ns1, но:
    allow-transfer { none; };     # ns2 зону дальше никому не отдаёт
    строк notify / also-notify НЕТ

include "/etc/named.transfer.key";
зоны ".", "localhost", "127.in-addr.arpa" - как на ns1

zone "lab1.org" IN {
    type slave;                               # slave = копия
    file "slaves/lab1.org.zone";              # файл создаст сам BIND
    masters { 192.168.122.55 key "transfer-key"; };   # откуда и с каким ключом брать
    allow-transfer { none; };
};
zone "122.168.192.in-addr.arpa" IN {
    type slave;
    file "slaves/192.168.122.zone";
    masters { 192.168.122.55 key "transfer-key"; };
    allow-transfer { none; };
};
logging { channel default_debug { file "data/named.run"; severity dynamic; }; };

#  ВАЖНО: файлы зон на ns2 руками НЕ создаём - они придут с ns1.

mkdir -p /var/named/slaves
chown named:named /var/named/slaves
#  ЗАЧЕМ: named должен иметь право ЗАПИСЫВАТЬ полученные копии зон.

named-checkconf -z /etc/named.conf
systemctl restart named
systemctl status named
#  В журнале должно быть: "transfer of 'lab1.org' from 192.168.122.55 ...
#  TSIG transfer-key ... Transfer completed" - ключ сработал.

ls -la /var/named/slaves/
#  появились lab1.org.zone и 192.168.122.zone (владелец named)
named-compilezone -f raw -F text -o - lab1.org /var/named/slaves/lab1.org.zone
#  ЗАЧЕМ: копии лежат в двоичном формате raw; так смотрим их как текст.
#  -f raw - формат входа, -F text - формат вывода, -o - - на экран.


---------------------------------------------------------------------
ШАГ 9. ПРОВЕРКИ (билет требует минимум ДВА способа)
---------------------------------------------------------------------
СПОСОБ 1: dig к каждому серверу напрямую         [ГДЕ: клиент или ns1]
dig @192.168.122.55 lab1.org SOA +short
dig @192.168.122.25 lab1.org SOA +short
#  serial ОДИНАКОВЫЙ = копия на ns2 свежая
dig @192.168.122.25 www.lab1.org +short          # 192.168.122.50
dig @192.168.122.25 -x 192.168.122.55 +short     # ns1.lab1.org.  (-x = PTR, обратная зона)
#  @ - какому серверу задать вопрос, +short - только ответ

СПОСОБ 2: nslookup и host
nslookup www.lab1.org 192.168.122.25
host www.lab1.org 192.168.122.55

СПОСОБ 3: синхронизация изменений                 [ГДЕ: ns1, потом ns2]
vim /var/named/lab1.org.zone
#   добавить строку:  newhost IN A 192.168.122.200
#   Serial 2024012002 -> 2024012003   (НЕ увеличишь - ns2 не заметит изменений)
systemctl reload named           # перечитать зону без остановки
dig @192.168.122.25 newhost.lab1.org +short     # 192.168.122.200 - пришло на ns2 само (NOTIFY)

СПОСОБ 4: защита трансфера
dig @192.168.122.55 lab1.org AXFR     # с клиента: "Transfer failed" - без ключа зону не отдают
rndc retransfer lab1.org              # на ns2: принудительный трансфер С ключом - без ошибок
tail /var/named/data/named.run        # в журнале: TSIG transfer-key, Transfer completed
rndc status                           # состояние сервера

СПОСОБ 5: через системный резолвер клиента        [ГДЕ: клиент]
vim /etc/resolv.conf
    nameserver 192.168.122.55
    nameserver 192.168.122.25
    options timeout:2 attempts:1
#  ЗАЧЕМ: два DNS-сервера; timeout:2 - через 2 секунды переходить к запасному.
dig www.lab1.org +short
nslookup www.lab1.org
ping -c 3 ns2.lab1.org


---------------------------------------------------------------------
ШАГ 10. ОТКАЗ PRIMARY (главная часть билета)
---------------------------------------------------------------------
systemctl stop named                                  # [ns1] имитируем поломку

dig @192.168.122.55 lab1.org SOA +short               # connection refused - ns1 не работает
#  "connection refused" = узел жив, служба остановлена.
#  (Если бы пропала сеть - было бы "timed out".)
dig @192.168.122.25 www.lab1.org +short               # 192.168.122.50 - ns2 отвечает из копии
dig @192.168.122.25 test.lab1.org +short              # 192.168.122.150

# с клиента, через резолвер (без @):
dig www.lab1.org +short          # сначала отказ от .55, затем ответ от .25
nslookup www.lab1.org
host test.lab1.org
#  ВЫВОД: резолвер сам переключился на ns2 - DNS продолжает работать.

systemctl start named                                 # [ns1] вернуть в работу


---------------------------------------------------------------------
ЧАСТЫЕ ОШИБКИ
---------------------------------------------------------------------
- Забыл точку в конце имени в зоне -> ns1.lab1.org.lab1.org.
- Поменял зону, но не увеличил Serial -> ns2 не обновляется.
- Ключ перенёс, но не сделал restorecon -> трансфер не идёт (SELinux).
- Не создал /var/named/slaves или владелец не named -> ns2 не может сохранить зону.
- Не открыл dns в firewalld -> трансфер по TCP 53 не проходит.
- Ошибся в имени файла служебной зоны -> named не стартует. Смотри ls /var/named.
- Диагностика: journalctl -u named -e   и   named-checkconf -z
